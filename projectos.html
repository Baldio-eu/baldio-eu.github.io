Código

Cola num HTML qualquer. Lê o baldio.json directamente (tem CORS aberto), não precisa de servidor.

<div id="baldio-projectos"></div>

<script>
fetch('https://baldio.eu/baldio.json')
  .then(response => response.json())
  .then(feed => {
    const escape = s => String(s ?? '').replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
    document.getElementById('baldio-projectos').innerHTML = feed.projectos
      .slice(0, 12)
      .map(p => `
        <a href="${escape(p.url)}" target="_blank" style="display:inline-block;width:260px;margin:8px;color:inherit;text-decoration:none">
          ${p.image
            ? `<img src="${escape(p.image)}" alt="" style="width:100%;aspect-ratio:16/9;object-fit:cover">`
            : `<div style="aspect-ratio:16/9;background:#2D4C36"></div>`}
          <strong>${escape(p.name)}</strong><br>
          <small>${escape(p.location)}</small>
        </a>`)
      .join('');
  });
</script>

Corre no teu servidor. Usa a API completa (/api/v1/projectos), que já traz banner e logo e aceita filtros como categoria, zona ou tag. Guarda em cache para não pedir à API a cada visita.

<?php
// Projectos do Baldio de uma categoria, com cache de 1 hora
$url   = 'https://baldio.eu/api/v1/projectos?categoria=quinta-agroecologia';
$cache = sys_get_temp_dir() . '/baldio-' . md5($url) . '.json';

if (is_file($cache) && time() - filemtime($cache) < 3600) {
    $json = file_get_contents($cache);
} else {
    $curl = curl_init($url);
    curl_setopt_array($curl, [CURLOPT_RETURNTRANSFER => true, CURLOPT_TIMEOUT => 10]);
    $json = curl_exec($curl);
    curl_close($curl);
    if ($json !== false) file_put_contents($cache, $json);
}

$data      = json_decode((string) $json, true) ?: [];
$projectos = array_slice($data['items'] ?? $data, 0, 12); // com filtros vem {items, meta}

foreach ($projectos as $p) {
    $imagem = $p['banner'] ?: $p['logo'];
    $link   = 'https://baldio.eu/projectos/' . $p['slug'];
    echo '<a href="' . htmlspecialchars($link) . '" target="_blank">';
    if ($imagem) echo '<img src="https://baldio.eu' . htmlspecialchars($imagem) . '" alt="" width="260">';
    echo '<strong>' . htmlspecialchars($p['name']) . '</strong> ';
    echo '<small>' . htmlspecialchars($p['location_label'] ?? '') . '</small></a>';
}
